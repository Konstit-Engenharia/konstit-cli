#compdef {{EXECUTABLE}}
# zsh completion for {{EXECUTABLE}}

{{NAME}}_takes_value() {
  case "$1:$2" in
{{TAKES_VALUE_CASES}}
    *) return 1 ;;
  esac
}

{{NAME}}_subcommand() {
  case "$1:$2" in
{{SUBCOMMAND_CASES}}
    *) return 1 ;;
  esac
}

{{NAME}}_candidates() {
  reply=()
  case "$1" in
{{CANDIDATE_CASES}}
  esac
}

{{NAME}}_complete_value() {
  local key="$1:$2" current="$3" prefix="$4"
  local -a values
  values=()
  case "$key" in
{{STATIC_VALUE_CASES}}
  esac
  if (( ${#values[@]} )); then
    compadd -Q -p "$prefix" -- "${values[@]}"
    return 0
  fi
  case "$key" in
{{VALUE_HINT_CASES}}
    *) return 1 ;;
  esac
}

{{NAME}}_allows_files() {
  case "$1" in
{{ALLOWS_FILES_CASES}}
    *) return 1 ;;
  esac
}

{{NAME}}() {
  local context="0" word child current previous option suffix prefix=""
  integer index expect_value=0 options_enabled=1 positional_handled=1
  local -a reply
  for ((index = 2; index < CURRENT; index++)); do
    word="${words[index]}"
    if (( expect_value )); then expect_value=0; continue; fi
    if [[ "$word" == "--" ]]; then options_enabled=0; continue; fi
    (( options_enabled )) || continue
    [[ "$word" == --*=* ]] && continue
    if {{NAME}}_takes_value "$context" "$word"; then expect_value=1; continue; fi
    child="$({{NAME}}_subcommand "$context" "$word")" && context="$child"
  done
  current="${words[CURRENT]}"
  previous="${words[CURRENT-1]}"
  if (( options_enabled )) && {{NAME}}_takes_value "$context" "$previous"; then
    {{NAME}}_complete_value "$context" "$previous" "$current" "" || _files
    return
  fi
  if (( options_enabled )) && [[ "$current" == --*=* ]]; then
    option="${current%%=*}"
    suffix="${current#*=}"
    prefix="$option="
    if {{NAME}}_takes_value "$context" "$option"; then
      {{NAME}}_complete_value "$context" "$option" "$suffix" "$prefix"
      return
    fi
  fi
  if [[ "$current" != -* ]]; then
    {{NAME}}_complete_value "$context" "@positional" "$current" ""
    positional_handled=$?
  fi
  if (( options_enabled )); then
    {{NAME}}_candidates "$context"
    (( ${#reply[@]} )) && compadd -Q -- "${reply[@]}"
  fi
  if (( positional_handled != 0 )) && [[ "$current" != -* ]] && {{NAME}}_allows_files "$context"; then
    _files
  fi
}

if [[ "$funcstack[1]" == {{NAME_QUOTED}} || "$funcstack[1]" == {{AUTOLOAD_NAME_QUOTED}} ]]; then
  {{NAME}} "$@"
else
  compdef {{NAME}} {{EXECUTABLE_QUOTED}}
fi
