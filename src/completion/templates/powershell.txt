# PowerShell completion for {{EXECUTABLE}}
Register-ArgumentCompleter -Native -CommandName {{EXECUTABLE_QUOTED}} -ScriptBlock {
  param($wordToComplete, $commandAst, $cursorPosition)

  $takesValue = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{TAKES_VALUE_CASES}}
      default { return $false }
    }
  }

  $getOptionSpec = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{OPTION_SPEC_CASES}}
      default { return $null }
    }
  }

  $getPositionalSpec = {
    param([string]$context, [int]$index)
    switch ("$context`:$index") {
{{POSITIONAL_SPEC_CASES}}
      default { return $null }
    }
  }

  $isLiteralPositional = {
    param([string]$context, [int]$index)
    switch ("$context`:$index") {
{{LITERAL_POSITIONAL_CASES}}
      default { return $false }
    }
  }

  $isDeclaredOption = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{DECLARED_OPTION_CASES}}
      default { return $false }
    }
  }

  $isBuiltinOption = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{BUILTIN_OPTION_CASES}}
      default { return $false }
    }
  }

  $getSubcommand = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{SUBCOMMAND_CASES}}
      default { return $null }
    }
  }

  $getValueMode = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{VALUE_MODE_CASES}}
      default { return $null }
    }
  }

  $getStaticValues = {
    param([string]$context, [string]$token)
    switch ("$context`:$token") {
{{STATIC_VALUE_CASES}}
      default { return @() }
    }
  }

  $getCandidates = {
    param([string]$context)
    switch ($context) {
{{CANDIDATE_CASES}}
      default { @() }
    }
  }

  $allowsFiles = {
    param([string]$context)
    switch ($context) {
{{ALLOWS_FILES_CASES}}
      default { return $false }
    }
  }

  $valueCount = {
    param([string]$value, [string]$delimiter)
    if ([string]::IsNullOrEmpty($delimiter)) { return 1 }
    return $value.Split([string[]]@($delimiter), [System.StringSplitOptions]::None).Count
  }
  $decodeHex = {
    param([string]$hex)
    if ($hex -eq '-' -or $hex -eq '00') { return '' }
    $bytes = New-Object byte[] ($hex.Length / 2)
    for ($byteIndex = 0; $byteIndex -lt $bytes.Length; $byteIndex++) {
      $bytes[$byteIndex] = [Convert]::ToByte($hex.Substring($byteIndex * 2, 2), 16)
    }
    return [Text.Encoding]::UTF8.GetString($bytes)
  }

  $context = '0'
  $pendingOption = $null
  $pendingSpec = $null
  $pendingConsumed = 0
  $pendingRawCount = 0
  $pendingDelimiter = ''
  $positionalIndex = 0
  $positionalRemaining = 0
  $positionalSpec = $null
  $positionalDelimiter = ''
  $positionalTerminator = ''
  $scanState = 'command'
  $tokens = @($commandAst.CommandElements | ForEach-Object { {{ARGV_WORD}} })
  $currentIncluded = $wordToComplete.Length -gt 0 -and $tokens.Count -gt 0
  $limit = $tokens.Count
  if ($currentIncluded) { $limit-- }
  for ($index = 1; $index -lt $limit; $index++) {
    $word = $tokens[$index]
    if ($scanState -eq 'option-values') {
      $pendingTerminator = & $decodeHex $pendingSpec.TerminatorHex
      if ($pendingTerminator.Length -gt 0 -and $word -ceq $pendingTerminator) {
        $scanState = 'command'
        continue
      }
      $stopPending = -not $pendingSpec.AllowHyphen -and $word -ne '-' -and $word.StartsWith('-') -and (-not $pendingSpec.AllowNegative -or $word -notmatch '^-(?:[0-9]|\.[0-9])')
      if (-not $stopPending -and $pendingConsumed -ge $pendingSpec.Minimum) {
        $stopPending = $null -ne (& $getSubcommand $context $word)
      }
      if (-not $stopPending) {
        $pendingConsumed += & $valueCount $word $pendingDelimiter
        $pendingRawCount++
        if (($pendingSpec.Maximum -ge 0 -and $pendingConsumed -ge $pendingSpec.Maximum) -or ($pendingSpec.SingleRaw -and $pendingRawCount -ge 1)) { $scanState = 'command' }
        continue
      }
      $scanState = 'command'
    }
    if ($scanState -eq 'command' -and (& $isLiteralPositional $context $positionalIndex)) { $scanState = 'literal' }
    $endOptions = $scanState -eq 'command' -and $word -eq '--'
    if ($endOptions) { $scanState = 'literal' }
    $consumeAsPositional = $false
    if ($scanState -eq 'command' -and $word -ne '-' -and $word.StartsWith('-')) {
      if ($positionalRemaining -eq 0) {
        $positionalSpec = & $getPositionalSpec $context $positionalIndex
        if ($null -ne $positionalSpec) {
          $positionalRemaining = $positionalSpec.Maximum
          $positionalDelimiter = & $decodeHex $positionalSpec.DelimiterHex
          $positionalTerminator = & $decodeHex $positionalSpec.TerminatorHex
        }
      }
      $declaredToken = if ($word -match '^(--[^=]+)=') { $Matches[1] } elseif ($word -match '^-[^-]') { $word.Substring(0, 2) } else { $word }
      $declared = (& $isBuiltinOption $context $word) -or (& $isDeclaredOption $context $declaredToken)
      if (-not $declared -and $null -ne $positionalSpec -and ($positionalSpec.AllowHyphen -or ($positionalSpec.AllowNegative -and $word -match '^-(?:[0-9]|\.[0-9])'))) { $consumeAsPositional = $true }
    }
    if ($scanState -eq 'command' -and -not $consumeAsPositional -and $word -match '^(--[^=]+)=') {
      $spec = & $getOptionSpec $context $Matches[1]
      if ($null -ne $spec) {
        $pendingSpec = $spec
        $pendingDelimiter = & $decodeHex $spec.DelimiterHex
        $pendingConsumed = & $valueCount ($word.Substring($word.IndexOf('=') + 1)) $pendingDelimiter
        $pendingRawCount = 1
        if (($spec.Maximum -lt 0 -or $pendingConsumed -lt $spec.Maximum) -and (-not $spec.SingleRaw)) {
          $pendingOption = $Matches[1]
          $scanState = 'option-values'
        }
      }
      continue
    }
    if ($scanState -eq 'command' -and -not $consumeAsPositional) {
      $spec = & $getOptionSpec $context $word
      if ($null -ne $spec) {
        if (-not $spec.RequireEquals -and $spec.Maximum -ne 0) {
          $pendingOption = $word
          $scanState = 'option-values'
          $pendingSpec = $spec
          $pendingDelimiter = & $decodeHex $spec.DelimiterHex
          $pendingConsumed = 0
          $pendingRawCount = 0
        }
        continue
      }
      if ($word -match '^-[^-]') {
        $cluster = $word.Substring(1)
        for ($shortIndex = 0; $shortIndex -lt $cluster.Length; $shortIndex++) {
          $short = "-$($cluster[$shortIndex])"
          $spec = & $getOptionSpec $context $short
          if ($null -eq $spec) {
            if (-not (& $isDeclaredOption $context $short)) { break }
            continue
          }
          $attached = $cluster.Substring($shortIndex + 1)
          $attachedPresent = $attached.Length -gt 0
          if ($attachedPresent -and $attached.StartsWith('=')) { $attached = $attached.Substring(1) }
          $pendingSpec = $spec
          $pendingDelimiter = & $decodeHex $spec.DelimiterHex
          $pendingConsumed = 0
          $pendingRawCount = 0
          if ($attachedPresent) {
            $pendingConsumed = & $valueCount $attached $pendingDelimiter
            $pendingRawCount = 1
          }
          if (($attachedPresent -or -not $spec.RequireEquals) -and ($spec.Maximum -lt 0 -or $pendingConsumed -lt $spec.Maximum) -and (-not $spec.SingleRaw -or $pendingRawCount -eq 0)) {
            $pendingOption = $short
            $scanState = 'option-values'
          }
          break
        }
        continue
      }
      $child = & $getSubcommand $context $word
      if ($null -ne $child) {
        $context = $child
        $positionalIndex = 0
        $positionalRemaining = 0
        continue
      }
    }
    if ($scanState -eq 'command' -and -not $consumeAsPositional -and $word -ne '-' -and $word.StartsWith('-')) { continue }
    if ($positionalRemaining -eq 0) {
      $positionalSpec = & $getPositionalSpec $context $positionalIndex
      if ($null -eq $positionalSpec) { continue }
      $positionalRemaining = $positionalSpec.Maximum
      $positionalDelimiter = & $decodeHex $positionalSpec.DelimiterHex
      $positionalTerminator = & $decodeHex $positionalSpec.TerminatorHex
    }
    if ($positionalTerminator.Length -gt 0 -and $word -eq $positionalTerminator) {
      $positionalIndex++
      $positionalRemaining = 0
      $positionalSpec = $null
      $positionalDelimiter = ''
      $positionalTerminator = ''
      continue
    }
    if ($endOptions) { continue }
    if ($positionalSpec.Trailing) { $scanState = 'literal' }
    if ($positionalRemaining -gt 0) {
      $positionalRemaining -= & $valueCount $word $positionalDelimiter
      if ($positionalRemaining -le 0) {
        if (-not $positionalSpec.Repeatable) { $positionalIndex++ }
        $positionalRemaining = 0
        $positionalSpec = $null
        $positionalDelimiter = ''
        $positionalTerminator = ''
      }
    }
  }

  $seen = [System.Collections.Generic.HashSet[string]]::new([System.StringComparer]::OrdinalIgnoreCase)
  $emit = {
    param([string]$value, [string]$description, [string]$prefix = '')
    $completion = "$prefix$value"
    if ($completion.StartsWith($wordToComplete, [System.StringComparison]::OrdinalIgnoreCase) -and $seen.Add($completion)) {
      if ([string]::IsNullOrEmpty($description)) { $description = $value }
      [System.Management.Automation.CompletionResult]::new($completion, $value, 'ParameterValue', $description)
    }
  }
  $completeValue = {
    param([string]$mode, [string]$token, [string]$partial, [string]$prefix = '')
    if ($mode -eq 'static') {
      foreach ($entry in @(& $getStaticValues $context $token)) {
        $value = [string]$entry.Value
        if ($value.StartsWith($partial, [System.StringComparison]::OrdinalIgnoreCase)) {
          & $emit $value $entry.Description $prefix
        }
      }
      return
    }
    if ([string]::IsNullOrEmpty($partial)) {
      $parent = '.'
      $leaf = ''
    } else {
      $parent = Split-Path -Path $partial -Parent
      $leaf = Split-Path -Path $partial -Leaf
      if ([string]::IsNullOrEmpty($parent)) { $parent = '.' }
    }
    Get-ChildItem -LiteralPath $parent -ErrorAction SilentlyContinue |
      Where-Object { ($mode -ne 'directory' -or $_.PSIsContainer) -and $_.Name.StartsWith($leaf, [System.StringComparison]::OrdinalIgnoreCase) } |
      ForEach-Object {
        $value = if ($parent -eq '.') { $_.Name } else { Join-Path $parent $_.Name }
        & $emit $value $value $prefix
      }
  }

  if ($scanState -eq 'option-values') {
    $pendingTerminator = & $decodeHex $pendingSpec.TerminatorHex
    $stopPending = ($pendingTerminator.Length -gt 0 -and $wordToComplete -ceq $pendingTerminator) -or (-not $pendingSpec.AllowHyphen -and $wordToComplete -ne '-' -and $wordToComplete.StartsWith('-') -and (-not $pendingSpec.AllowNegative -or $wordToComplete -notmatch '^-(?:[0-9]|\.[0-9])'))
    if (-not $stopPending -and $pendingConsumed -ge $pendingSpec.Minimum) {
      $stopPending = $null -ne (& $getSubcommand $context $wordToComplete)
    }
    if (-not $stopPending) {
      $mode = & $getValueMode $context $pendingOption
      if ($null -eq $mode) { $mode = 'file' }
      & $completeValue $mode $pendingOption $wordToComplete
      return
    }
    $scanState = 'command'
  }
  if ($scanState -eq 'command' -and (& $isLiteralPositional $context $positionalIndex)) { $scanState = 'literal' }
  $consumeAsPositional = $scanState -eq 'literal' -or -not $wordToComplete.StartsWith('-')
  if (-not $consumeAsPositional -and $wordToComplete -ne '--') {
    if ($positionalRemaining -eq 0) { $positionalSpec = & $getPositionalSpec $context $positionalIndex }
    $declaredToken = if ($wordToComplete -match '^(--[^=]+)=') { $Matches[1] } elseif ($wordToComplete -match '^-[^-]') { $wordToComplete.Substring(0, 2) } else { $wordToComplete }
    $declared = (& $isBuiltinOption $context $wordToComplete) -or (& $isDeclaredOption $context $declaredToken)
    if (-not $declared -and $null -ne $positionalSpec -and ($positionalSpec.AllowHyphen -or ($positionalSpec.AllowNegative -and $wordToComplete -match '^-(?:[0-9]|\.[0-9])'))) { $consumeAsPositional = $true }
  }
  if ($scanState -eq 'command' -and $wordToComplete -match '^(--[^=]+)=(.*)$' -and (& $takesValue $context $Matches[1])) {
    $option = $Matches[1]
    $partial = $Matches[2]
    $mode = & $getValueMode $context $option
    if ($null -eq $mode) { $mode = 'file' }
    & $completeValue $mode $option $partial "$option="
    return
  }
  if ($scanState -eq 'command' -and $wordToComplete -match '^-[^-]') {
    $cluster = $wordToComplete.Substring(1)
    for ($shortIndex = 0; $shortIndex -lt $cluster.Length; $shortIndex++) {
      $short = "-$($cluster[$shortIndex])"
      if (-not (& $takesValue $context $short)) {
        if (-not (& $isDeclaredOption $context $short)) { break }
        continue
      }
      $attached = $cluster.Substring($shortIndex + 1)
      $attachedPresent = $attached.Length -gt 0
      if ($attachedPresent -and $attached.StartsWith('=')) { $attached = $attached.Substring(1) }
      if ($attachedPresent) {
        $prefix = $wordToComplete.Substring(0, $shortIndex + 2)
        if ($cluster.Substring($shortIndex + 1).StartsWith('=')) { $prefix += '=' }
        $mode = & $getValueMode $context $short
        if ($null -eq $mode) { $mode = 'file' }
        & $completeValue $mode $short $attached $prefix
        return
      }
      break
    }
  }
  if ($consumeAsPositional) {
    $positional = "@positional`:$positionalIndex"
    $mode = & $getValueMode $context $positional
    if ($null -ne $mode) { & $completeValue $mode $positional $wordToComplete }
    elseif (& $allowsFiles "$context`:$positionalIndex") { & $completeValue 'file' $positional $wordToComplete }
  }
  if ($scanState -eq 'command' -and -not $consumeAsPositional) {
    foreach ($candidate in @(& $getCandidates $context)) {
      & $emit $candidate.Value $candidate.Description
    }
  }
}
